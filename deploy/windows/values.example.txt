# Answers for an install with no one present: install.ps1 -ZipPath <zip> -ValuesFile <this file, filled in> -NonInteractive
#
# Copy this file, replace every CHANGE_ME value, run the install, then DELETE your copy: it holds the first Admin's password.
# One NAME=value per line. Lines that start with # are ignored. Never commit a filled-in copy.
#
# What is not here is invented by the install: the session secret (AUTH_SECRET) and the two database passwords.
# Add AUTH_SECRET=, DB_OWNER_PASSWORD= or DB_APP_PASSWORD= only if you must choose them yourself.

# The address people will type (a scheme and a host, no path). On a server people use, the portal sits behind a TLS reverse proxy and you run the install with -WithProxy (README, "Reverse proxy"):
# then this must be the proxy's https address, with no port, and the install refuses anything else.
APP_ORIGIN=https://CHANGE_ME_SERVER_NAME

# The first Admin. The password: 10 to 128 characters, with a digit and a symbol. They choose a new one at first sign-in.
ADMIN_EMAIL=CHANGE_ME@example.test
ADMIN_INITIAL_PASSWORD=CHANGE_ME_FIRST_PASSWORD

# The age PUBLIC key that every backup is encrypted to (it starts with age1). Make the key pair with age-keygen on YOUR OWN computer, keep the private key
# (AGE-SECRET-KEY-...) there, in two places, and give the install only the public key. The private key must never be put on this server or in this file:
# whoever has it can read every backup, and without it no backup can be restored.
BACKUP_AGE_RECIPIENT=CHANGE_ME_AGE_PUBLIC_KEY

# The password of the PostgreSQL administrator (the one you typed when you installed PostgreSQL). Used by the install only, never stored.
PG_SUPERUSER_PASSWORD=CHANGE_ME_POSTGRES_ADMIN_PASSWORD

# Mail. none = no mail yet. Other modes: password, oauth2-client-credentials (then also SMTP_USER and the fields for the mode).
SMTP_AUTH_MODE=none
SMTP_HOST=localhost
SMTP_PORT=1025
