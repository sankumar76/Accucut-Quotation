async function ownerClient(): Promise<Client> {
  const url = process.env['MIGRATE_DATABASE_URL'];
  if (url === undefined || !url.includes('localhost:55523')) throw new Error('The QA probe runs only against its own throwaway database on port 55523.');
  const client = new Client({ connectionString: url });
  await client.connect();
  return client;
}

async function pairInDatabase(): Promise<Record<string, string>> {
  const client = await ownerClient();
  try {
    const { rows } = await client.query<{ currency: string; v: string }>('SELECT currency, inr_value::text AS v FROM exchange_rates');
    return Object.fromEntries(rows.map((row) => [row.currency, row.v]));
  } finally {
    await client.end();
  }
}

async function exchangeAudit(): Promise<Array<{ event: string; object_type: string; object_id: string; before: unknown; after: unknown; actor_label: string }>> {
  const client = await ownerClient();
  try {
    const { rows } = await client.query(
      "SELECT event, object_type, object_id::text, before, after, actor_label FROM audit_log WHERE object_type = 'exchange_rate' ORDER BY created_at, id",
    );
    return rows;
  } finally {
    await client.end();
  }
}

async function open(page: Page): Promise<void> {
  await page.goto('/masters/materials');
  await expect(page.getByRole('heading', { level: 1, name: 'Materials and lists' })).toBeVisible();
  await waitUntilHydrated(page, 'main button[data-action="add"]');
  await page.getByRole('tab', { name: 'Materials' }).focus();
  await page.keyboard.press('ArrowRight');
  await expect(page.getByLabel(LABEL_USD)).toBeVisible();
}

const usd = (page: Page) => page.getByLabel(LABEL_USD);
const eur = (page: Page) => page.getByLabel(LABEL_EUR);
const save = (page: Page) => page.getByRole('button', { name: SAVE });

/** Types into the focused field by keyboard: select all, then type. */
async function typeOver(page: Page, text: string): Promise<void> {
  await page.keyboard.press('ControlOrMeta+A');
  await page.keyboard.type(text);
}

/** Keyboard only, from the first field: USD, Tab, EUR, Tab, Save, Enter. Returns the PUT response. */
async function keyboardSave(page: Page, pair: { usd: string; eur: string }) {
  await usd(page).focus();
  await typeOver(page, pair.usd);
  await page.keyboard.press('Tab');
  await expect(eur(page)).toBeFocused();
  await typeOver(page, pair.eur);
  await page.keyboard.press('Tab');
  await expect(save(page)).toBeFocused();
  const answered = page.waitForResponse((r) => r.request().method() === 'PUT' && r.url().endsWith('/api/v1/exchange-rates'));
  await page.keyboard.press('Enter');
  return answered;
}

/** Both fields refused by keyboard: USD "abc", EUR "-1", Save. Focus ends on USD (first message). */
async function refuseBoth(page: Page): Promise<void> {
  await usd(page).focus();
  await typeOver(page, 'abc');
  await page.keyboard.press('Tab');
  await typeOver(page, '-1');
  await page.keyboard.press('Tab');
  await page.keyboard.press('Enter');
  await expect(page.getByText(NUMBER_MESSAGE)).toHaveCount(2);
  await expect(usd(page)).toBeFocused();
}

test('D3-7 QLT-11: a keyboard-only save of a pair other than the canary pair, through the real route, stores both with one stamp and writes one master.edited entry; then the canary pair is restored', async ({ page }) => {
  await ensureCanaryExchangeRates();
  expect(await pairInDatabase()).toEqual({ USD: `${CANARY.usd}00`, EUR: `${CANARY.eur}00` });
  const entriesBefore = (await exchangeAudit()).length;

  await open(page);
  await expect(usd(page)).toHaveValue(CANARY.usd);
  const first = await keyboardSave(page, OTHER_PAIR);
  expect(first.status()).toBe(200);
  const body = (await first.json()) as { usd: string; eur: string; updatedBy: string; updatedAt: string };
  expect(body).toMatchObject({ usd: OTHER_PAIR.usd, eur: OTHER_PAIR.eur, updatedBy: E2E_ADMIN.name });
  await expect(page.getByRole('status').filter({ hasText: SAVED })).toBeVisible();
  await expect(save(page)).toBeFocused();

  const client = await ownerClient();
  try {
    const { rows } = await client.query<{ currency: string; v: string; by: string; at: Date }>(
      'SELECT currency, inr_value::text AS v, updated_by::text AS by, updated_at AS at FROM exchange_rates ORDER BY currency',
    );
    expect(rows.map((row) => [row.currency, row.v])).toEqual([['EUR', `${OTHER_PAIR.eur}00`], ['USD', `${OTHER_PAIR.usd}00`]]);
    expect(rows[0]?.at.getTime()).toBe(rows[1]?.at.getTime());
    expect(rows[0]?.by).toBe(rows[1]?.by);
    expect(rows[0]?.at.toISOString()).toBe(body.updatedAt);
    console.log('QLT11 rows', JSON.stringify(rows.map((r) => ({ currency: r.currency, value: r.v, updated_by: r.by, updated_at: r.at.toISOString() }))));
  } finally {
    await client.end();
  }
  const afterFirst = await exchangeAudit();
  expect(afterFirst).toHaveLength(entriesBefore + 1);
  console.log('QLT11 entry 1', JSON.stringify(afterFirst.at(-1)));
  expect(afterFirst.at(-1)).toMatchObject({
    event: 'master.edited',
    object_type: 'exchange_rate',
    object_id: FIXED_OBJECT_ID,
    actor_label: E2E_ADMIN.name,
    before: { usd: CANARY.usd, eur: CANARY.eur },
    after: OTHER_PAIR,
  });
  // The stamp line on screen is that instant in IST, worked out by hand.
  const shifted = new Date(new Date(body.updatedAt).getTime() + 330 * 60_000);
  const months = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];
  const two = (n: number) => String(n).padStart(2, '0');
  const ist = `${two(shifted.getUTCDate())} ${months[shifted.getUTCMonth()]} ${shifted.getUTCFullYear()}, ${two(shifted.getUTCHours())}:${two(shifted.getUTCMinutes())} IST`;
  await expect(page.getByText(`Last updated by ${E2E_ADMIN.name}, ${ist}.`)).toBeVisible();

  // A second press with nothing changed: Nothing to save, no request, no entry.
  let puts = 0;
  page.on('request', (r) => {
    if (r.method() === 'PUT') puts += 1;
  });
  await save(page).focus();
  await page.keyboard.press('Enter');
  await expect(page.getByRole('status').filter({ hasText: NOTHING_TO_SAVE })).toBeVisible();
  expect(puts).toBe(0);

  // Restore the canary pair, keyboard only, and check what the scan needs.
  const second = await keyboardSave(page, CANARY);
  expect(second.status()).toBe(200);
  await expect(page.getByRole('status').filter({ hasText: SAVED })).toBeVisible();
  expect(await pairInDatabase()).toEqual({ USD: `${CANARY.usd}00`, EUR: `${CANARY.eur}00` });
  const afterSecond = await exchangeAudit();
  expect(afterSecond).toHaveLength(entriesBefore + 2);
  console.log('QLT11 entry 2 (restore)', JSON.stringify(afterSecond.at(-1)));
  console.log('QLT11 entries before/after', entriesBefore, afterSecond.length, 'pair now', JSON.stringify(await pairInDatabase()));
  expect(afterSecond.at(-1)).toMatchObject({ event: 'master.edited', object_id: FIXED_OBJECT_ID, before: OTHER_PAIR, after: { usd: CANARY.usd, eur: CANARY.eur } });
});

test('D3-8: the five sentences of ux-spec v0.21, in the browser', async ({ page }) => {
  // Loading (a slow read), then the form.
  let release!: () => void;
  const gate = new Promise<void>((resolve) => {
    release = resolve;
  });
  let held = true;
  await page.route('**/api/v1/exchange-rates', async (route) => {
    if (route.request().method() === 'GET' && held) {
      await gate;
      held = false;
    }
    await route.fallback();
  });
  await page.goto('/masters/materials');
  await waitUntilHydrated(page, 'main button[data-action="add"]');
  await page.getByRole('tab', { name: 'Exchange rates' }).click();
  await expect(page.getByText(LOADING)).toBeVisible();
  release();
  await expect(usd(page)).toBeVisible();
  await page.unroute('**/api/v1/exchange-rates');

  // Failed: the title is bold, then the sentence; what was typed stays; no "Nothing was changed."
  await page.route('**/api/v1/exchange-rates', (route) =>
    route.request().method() === 'PUT'
      ? route.fulfill({ status: 503, contentType: 'application/problem+json', body: JSON.stringify({ type: 'about:blank', title: 'busy', status: 503, code: 'DATABASE_BUSY', requestId: 'req-0a0b0c0d' }) })
      : route.fallback(),
  );
  await usd(page).fill('99.37');
  await save(page).click();
  await expect(page.getByText(FAILED_TITLE)).toBeVisible();
  await expect(page.getByText(FAILED_BODY)).toBeVisible();
  await expect(page.getByText('Nothing was changed.')).toHaveCount(0);
  await expect(page.getByRole('alert').filter({ hasText: FAILED_TITLE })).toBeVisible();
  await expect(usd(page)).toHaveValue('99.37');
  await page.unroute('**/api/v1/exchange-rates');

  // Load error without a Saved notice: title takes focus, sentence says "Nothing was changed."
  await page.route('**/api/v1/exchange-rates', (route) => (route.request().method() === 'GET' ? route.fulfill({ status: 503, contentType: 'application/problem+json', body: JSON.stringify({ type: 'about:blank', title: 'busy', status: 503, code: 'DATABASE_BUSY', requestId: 'req-0a0b0c0d' }) }) : route.fallback()));
  await page.goto('/masters/materials');
  await waitUntilHydrated(page, 'main button[data-action="add"]');
  await page.getByRole('tab', { name: 'Exchange rates' }).click();
  await expect(page.getByText(LOAD_ERROR_TITLE)).toBeVisible();
  await expect(page.getByText(LOAD_ERROR_BODY)).toBeVisible();
  await expect(page.getByRole('heading', { name: LOAD_ERROR_TITLE })).toBeFocused();
  await expect(page.getByRole('button', { name: 'Try again' })).toBeVisible();
  await page.unroute('**/api/v1/exchange-rates');
});

test('D3-8b: Error with the Saved notice on the page says the rates were saved but cannot be shown; focus on the error title and stays', async ({ page, request }) => {
  await ensureCanaryExchangeRates();
  const real = (await (await request.get(API)).json()) as Record<string, unknown>;
  // The screen is told another pair is stored, so sending the canary pair is a change on screen and a no-op write on the server; the re-read then fails.
  let putSeen = false;
  await page.route('**/api/v1/exchange-rates', (route) => {
    if (route.request().method() !== 'GET') {
      putSeen = true;
      return route.fallback();
    }
    return !putSeen
      ? route.fulfill({ status: 200, contentType: 'application/json', body: JSON.stringify({ ...real, usd: OTHER_PAIR.usd, eur: OTHER_PAIR.eur }) })
      : route.fulfill({ status: 503, contentType: 'application/problem+json', body: JSON.stringify({ type: 'about:blank', title: 'busy', status: 503, code: 'DATABASE_BUSY', requestId: 'req-0a0b0c0d' }) });
  });
  await open(page);
  const answered = await keyboardSave(page, CANARY);
  expect(answered.status()).toBe(200);
  await expect(page.getByText(SAVED_NOT_SHOWN)).toBeVisible();
  await expect(page.getByRole('heading', { name: LOAD_ERROR_TITLE })).toBeFocused();
  await expect(page.getByText('Nothing was changed.')).toHaveCount(0);
  await page.waitForTimeout(500);
  await expect(page.getByRole('heading', { name: LOAD_ERROR_TITLE })).toBeFocused();
  expect(await pairInDatabase()).toEqual({ USD: `${CANARY.usd}00`, EUR: `${CANARY.eur}00` });
});

test('D3-9 AC-6: a Lead and an Engineer have no way to the tab and no rate over the wire; signed out is refused', async ({ browser }) => {
  test.setTimeout(180_000);
  await ensureCanaryExchangeRates();
  const lead = await browser.newContext({ baseURL: DEVELOPMENT_URL, storageState: SIGNED_IN_STATE });
  try {
    const page = await lead.newPage();

    expect(answer?.status()).toBe(403);
    await expect(page.getByRole('tab', { name: 'Exchange rates' })).toHaveCount(0);

    expect(seen.join(' ')).not.toMatch(/838[.,]?37|839[.,]?73|inrValue|"usd"|"eur"/);
  } finally {
    await lead.close();

    await signInThroughThePage(page, E2E_ENGINEER.email, E2E_PASSWORD);
    await expect(page).toHaveURL(/queue$/, { timeout: 120_000 });
    const answer = await page.goto('/masters/materials');
    expect(answer?.status()).toBe(403);
    await expect(page.getByRole('tab', { name: 'Exchange rates' })).toHaveCount(0);
    const seen: string[] = [await page.content()];
    for (const method of ['get', 'put'] as const) {
      const result = await page.request[method](API, { headers: { origin: DEVELOPMENT_URL, 'content-type': 'application/json' }, ...(method === 'put' ? { data: CANARY } : {}) });

    }
    expect(seen.join(' ')).not.toMatch(/838[.,]?37|839[.,]?73|inrValue|"usd"|"eur"/);
  } finally {
    await engineer.close();
  }
  const anonymous = await browser.newContext({ baseURL: DEVELOPMENT_URL, storageState: NOBODY });
  try {
    expect((await anonymous.request.get(API)).status()).toBe(401);
    expect((await anonymous.request.put(API, { headers: { origin: DEVELOPMENT_URL, 'content-type': 'application/json' }, data: CANARY })).status()).toBe(401);

  }
  expect(await pairInDatabase()).toEqual({ USD: `${CANARY.usd}00`, EUR: `${CANARY.eur}00` });
});

test('E-1 exploratory: a stale second tab saves the values another tab already saved: 200, "Exchange rates saved.", and no second audit entry (QLT-06 a)', async ({ browser }) => {
  await ensureCanaryExchangeRates();
  const tabA = await browser.newContext({ baseURL: DEVELOPMENT_URL, storageState: ADMIN_STATE });

  expect(await pairInDatabase()).toEqual({ USD: `${CANARY.usd}00`, EUR: `${CANARY.eur}00` });
